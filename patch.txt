local PUMP_FRAME_NAME = "pump_configuration_frame"
local PROJECTOR_FRAME_NAME = "projector_configuration_frame"

local function is_valid_frame_name(name)
    return name == PUMP_FRAME_NAME or name == PROJECTOR_FRAME_NAME
end

local function notify_change(unit_number, is_projector)
    local entity = nil
    if is_projector then
        entity = (storage.active_projectors and storage.active_projectors[unit_number]) or
                 (storage.ghost_devices and storage.ghost_devices[unit_number])
    else
        entity = (storage.active_pumps and storage.active_pumps[unit_number]) or
                 (storage.ghost_devices and storage.ghost_devices[unit_number])
    end
    if entity and entity.valid then
        active_device_scanner.notify_settings_changed(entity)
    end
end

function pump_gui.close(player)
    if not (player and player.valid) then return end
    if player.gui.screen[PUMP_FRAME_NAME] then
        player.gui.screen[PUMP_FRAME_NAME].destroy()
    end
    if player.gui.screen[PROJECTOR_FRAME_NAME] then
        player.gui.screen[PROJECTOR_FRAME_NAME].destroy()
    end
    if player.opened and player.opened.valid and is_valid_frame_name(player.opened.name) then
        player.opened = nil
    end
end

function pump_gui.open(player, entity)
    if not (player and player.valid and entity and entity.valid) then return end

    pump_gui.close(player)

    local is_ghost = (entity.name == "entity-ghost")
    local real_name = is_ghost and entity.ghost_name or entity.name
    local is_projector = (real_name == "pneumatic-projector")

    local dev_id = is_projector and projector_settings.get_device_id(entity) or pump_settings.get_device_id(entity)
    dev_id = dev_id or entity.unit_number

    local settings = is_projector and projector_settings.get(dev_id, entity) or pump_settings.get(dev_id)
    if not settings then return end

    if is_ghost then
        storage.ghost_devices = storage.ghost_devices or {}
        storage.ghost_devices[dev_id] = entity
    end

    local frame_name = is_projector and PROJECTOR_FRAME_NAME or PUMP_FRAME_NAME
    local main_frame = gui_components.create_relative_window(player, nil, frame_name)
    if not main_frame then return end

    local base_tags = { unit_number = dev_id, is_projector = is_projector }

    -- Titlebar Header
    local title = is_projector and "Electromagnetic Projector Configuration" or "Pneumatic Pump Configuration"
    local close_btn_name = is_projector and "projector_close_button" or "pump_close_button"
    gui_components.add_header(main_frame, title, close_btn_name, base_tags)

    -- Global Wire Channel Toggles
    gui_components.add_wire_channel_toggles(main_frame, {
        read_red_name = is_projector and "projector_read_red" or "pump_read_red",
        read_green_name = is_projector and "projector_read_green" or "pump_read_green",
        read_red_state = settings.read_red ~= false,
        read_green_state = settings.read_green ~= false,
        tags = base_tags
    })

    -- Configuration Card
    local card_frame = gui_components.add_card_frame(main_frame, "vertical")

    -- Enable Checkbox Header
    local header_flow = card_frame.add{ type = "flow", direction = "horizontal" }
    header_flow.style.vertical_align = "center"
    header_flow.add{
        type = "checkbox",
        name = is_projector and "projector_enable" or "pump_enable",
        caption = is_projector and "Enable Projector" or "Enable Pump",
        state = settings.enabled,
        tags = base_tags
    }

    -- Circuit Network Enable Row
    local cond = settings.enable_condition or { first_signal = nil, comparator = "=", constant = 0 }
    gui_components.add_circuit_condition_panel(card_frame, {
        checkbox_name = is_projector and "projector_use_circuit_enable" or "pump_use_circuit_enable",
        checkbox_caption = "Circuit Enable",
        checkbox_state = settings.use_circuit_enable or false,
        signal = cond.first_signal,
        comparator = cond.comparator or "=",
        constant = cond.constant or 0,
        signal_button_name = is_projector and "projector_circuit_signal" or "pump_circuit_signal",
        comparator_dropdown_name = is_projector and "projector_circuit_comparator" or "pump_circuit_comparator",
        constant_textfield_name = is_projector and "projector_circuit_constant" or "pump_circuit_constant",
        tags = base_tags
    })

    player.opened = main_frame
end

pump_gui.open_projector = pump_gui.open
pump_gui.close_projector = pump_gui.close

local function on_gui_click(event)
    local element = event.element
    if not (element and element.valid) then return end
    if element.name == "pump_close_button" or element.name == "projector_close_button" then
        local player = game.get_player(event.player_index)
        if player and player.valid then
            pump_gui.close(player)
        end
    end
end

local function on_gui_checked_state_changed(event)
    local element = event.element
    if not (element and element.valid) then return end
    local tags = element.tags
    if not (tags and tags.unit_number) then return end

    local is_projector = tags.is_projector == true
    local settings = is_projector and projector_settings.get(tags.unit_number) or pump_settings.get(tags.unit_number)
    if not settings then return end

    local name = element.name
    if name == "pump_read_red" or name == "projector_read_red" then
        settings.read_red = element.state
    elseif name == "pump_read_green" or name == "projector_read_green" then
        settings.read_green = element.state
    elseif name == "pump_enable" or name == "projector_enable" then
        settings.enabled = element.state
    elseif name == "pump_use_circuit_enable" or name == "projector_use_circuit_enable" then
        settings.use_circuit_enable = element.state
    end

    notify_change(tags.unit_number, is_projector)
end

local function on_gui_elem_changed(event)
    local element = event.element
    if not (element and element.valid) then return end
    local tags = element.tags
    if not (tags and tags.unit_number) then return end

    local is_projector = tags.is_projector == true
    local settings = is_projector and projector_settings.get(tags.unit_number) or pump_settings.get(tags.unit_number)
    if not settings then return end

    local name = element.name
    if name == "pump_circuit_signal" or name == "projector_circuit_signal" then
        if not settings.enable_condition then
            settings.enable_condition = { first_signal = nil, comparator = "=", constant = 0 }
        end
        settings.enable_condition.first_signal = element.elem_value
        notify_change(tags.unit_number, is_projector)
    end
end

local function on_gui_text_changed(event)

    if not (element and element.valid) then return end
    local tags = element.tags
    if not (tags and tags.unit_number) then return end

    local is_projector = tags.is_projector == true
